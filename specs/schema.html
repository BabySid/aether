▼ Level 6 — Primitives
Name = string /^[a-z0-9]([-a-z0-9]*[a-z0-9])?$/ [1..63] # DNS-1123 compliant name
Duration = string /^([0-9]+d)?([0-9]+h)?([0-9]+m)?([0-9]+s)?$/ # e.g. "30s", "5m", "2h", "1d", "1h30m"
Phase = "Created" # persisted in store, engine not yet committed | "Ready" # engine committed to execute (leaf dispatched / container scope opened) | "Running" # worker confirmed start (OnTaskStarted fired) | "Suspended" # executor returned ExecCodeSuspended, awaiting Resume() | "Succeeded" # success | "Failed" # business failure (executor returned Failed) | "Error" # system/infrastructure error | "Skipped" # when=false (condition not met) | "Timeout" # execution timed out | "Cancelled" # terminated via Cancel API; not retried
AnyValue = string | number | boolean | object | array
